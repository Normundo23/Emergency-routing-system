3.2 Implemented Test Code (Snippet)

    def test_TC01_high_risk_scenario(model_and_data):
        """
        Scenario: Rush Hour (8am) + Rain + Highway + High Traffic
        Expected: High Risk (1)
        """
        clf, _, _ = model_and_data
        # Input: [Time:8, Weather:1, Road:0, Vol:1200]
        incident = np.array([[8, 1, 0, 1200]]) 
        prediction = clf.predict(incident)[0]
        assert prediction == 1 # Expected: High Risk

    def test_TC02_low_risk_scenario(model_and_data):
        """
        Scenario: Night (3am) + Clear + City + Low Traffic
        Expected: Low Risk (0)
        """
        clf, _, _ = model_and_data
        # Input: [Time:3, Weather:0, Road:1, Vol:50]
        incident = np.array([[3, 0, 1, 50]]) 
        prediction = clf.predict(incident)[0]
        assert prediction == 0 # Expected: Low Risk
