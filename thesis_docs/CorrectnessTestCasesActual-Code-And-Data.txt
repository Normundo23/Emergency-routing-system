2. PYTEST WITH ACTUAL PROJECT DATA
---------------------------------------------------------------------
import pytest
import pandas as pd
import numpy as np
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

@pytest.fixture(scope="module")
def model_and_data():
    # Load raw Emergency Incident CSV
    df = pd.read_csv("emergency_incidents.csv")
    X = df[["TimeOfDay", "WeatherCondition", "RoadType", "TrafficVolume"]].values
    y = df["RiskLevel"].values
    
    # Train-test split (30% test)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)
    
    # Train actual project Decision Tree
    clf = DecisionTreeClassifier(random_state=42)
    clf.fit(X_train, y_train)
    return clf, X_test, y_test

# --- Single Incident Test Cases ---
def test_TC01_rush_hour_rain_high_risk(model_and_data):
    clf, _, _ = model_and_data
    # 8AM + Rain + Highway + 1200 Vehicles
    incident = np.array([[8, 1, 0, 1200]]) 
    prediction = clf.predict(incident)[0]
    assert prediction == 1  # Expected: High Risk

def test_TC02_night_clear_low_risk(model_and_data):
    clf, _, _ = model_and_data
    # 3AM + Clear + City + 50 Vehicles
    incident = np.array([[3, 0, 1, 50]])
    prediction = clf.predict(incident)[0]
    assert prediction == 0  # Expected: Low Risk

# --- Dataset-level Evaluation ---
def test_TC09_accuracy_benchmark(model_and_data):
    clf, X_test, y_test = model_and_data
    y_pred = clf.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    assert acc >= 0.75
